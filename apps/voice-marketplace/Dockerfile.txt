#!/usr/bin/env python3
import sys
import httpx
import os

def health_check():
    try:
        port = os.getenv('PORT', '8006')
        response = httpx.get(f'http://localhost:{port}/health', timeout=10)
        if response.status_code == 200:
            data = response.json()
            if data.get('status') == 'healthy':
                print("✅ Voice Marketplace service is healthy")
                return 0
        print(f"❌ Health check failed: {response.status_code}")
        return 1
    except Exception as e:
        print(f"❌ Health check error: {e}")
        return 1

if __name__ == '__main__':
    sys.exit(health_check())
EOF

RUN chmod +x /app/healthcheck.py

import time
import psycopg2
import os
from urllib.parse import urlparse

url = urlparse(os.getenv('DATABASE_URL'))
max_retries = 30
retry_count = 0

while retry_count < max_retries:
    try:
        conn = psycopg2.connect(
            host=url.hostname,
            port=url.port,
            user=url.username, 
            password=url.password,
            database=url.path[1:]
        )
        conn.close()
        print('✅ Database connection successful')
        break
    except Exception as e:
        retry_count += 1
        print(f'⏳ Database not ready, retrying... ({retry_count}/{max_retries})')
        time.sleep(2)
else:
    print('❌ Database connection failed after all retries')

# apps/voice-marketplace/scripts/build.sh
#!/bin/bash

# 🎭 Vocelio Voice Marketplace - Docker Build Script

set -e

# Configuration
SERVICE_NAME="voice-marketplace"
REGISTRY="${DOCKER_REGISTRY:-vocelio}"
VERSION="${VERSION:-$(git describe --tags --always)}"
BUILD_DATE=$(date -u +'%Y-%m-%dT%H:%M:%SZ')
PLATFORM="${PLATFORM:-linux/amd64,linux/arm64}"

echo "🎭 Building Vocelio Voice Marketplace Docker Images"
echo "📦 Service: $SERVICE_NAME"
echo "🏷️ Version: $VERSION"
echo "📅 Build Date: $BUILD_DATE"
echo "🏗️ Platform: $PLATFORM"

# Build development image
echo "🔨 Building development image..."
docker build \
  --target development \
  --build-arg BUILD_DATE="$BUILD_DATE" \
  --build-arg VERSION="$VERSION" \
  --tag "$REGISTRY/$SERVICE_NAME:dev" \
  --tag "$REGISTRY/$SERVICE_NAME:dev-$VERSION" \
  .

# Build production image
echo "🔨 Building production image..."
docker build \
  --target production \
  --build-arg BUILD_DATE="$BUILD_DATE" \
  --build-arg VERSION="$VERSION" \
  --tag "$REGISTRY/$SERVICE_NAME:latest" \
  --tag "$REGISTRY/$SERVICE_NAME:$VERSION" \
  .

# Build multi-arch image (optional)
if command -v docker buildx >/dev/null 2>&1; then
  echo "🔨 Building multi-architecture image..."
  docker buildx build \
    --platform "$PLATFORM" \
    --target multi-arch \
    --build-arg BUILD_DATE="$BUILD_DATE" \
    --build-arg VERSION="$VERSION" \
    --tag "$REGISTRY/$SERVICE_NAME:multi-arch" \
    --push \
    . || echo "⚠️ Multi-arch build failed (buildx not configured)"
fi

echo "✅ Build completed successfully!"
echo "🏷️ Images tagged:"
echo "  - $REGISTRY/$SERVICE_NAME:dev"
echo "  - $REGISTRY/$SERVICE_NAME:latest"
echo "  - $REGISTRY/$SERVICE_NAME:$VERSION"

# Run basic tests on the image
echo "🧪 Running basic image tests..."
docker run --rm "$REGISTRY/$SERVICE_NAME:latest" python -c "
import sys
print(f'Python version: {sys.version}')
import fastapi, sqlalchemy, stripe
print('✅ All dependencies imported successfully')
"

echo "🎉 Voice Marketplace Docker build complete!"

#!/bin/bash

# 🚀 Vocelio Voice Marketplace - Deployment Script

set -e

ENVIRONMENT="${ENVIRONMENT:-development}"
SERVICE_NAME="voice-marketplace"
REGISTRY="${DOCKER_REGISTRY:-vocelio}"
VERSION="${VERSION:-latest}"

echo "🚀 Deploying Vocelio Voice Marketplace"
echo "🌍 Environment: $ENVIRONMENT"
echo "🏷️ Version: $VERSION"

case $ENVIRONMENT in
  "development")
    echo "🔧 Starting development environment..."
    docker-compose -f docker-compose.yml up -d
    ;;
  "production")
    echo "🏭 Deploying to production..."
    docker-compose -f docker-compose.yml -f docker-compose.prod.yml up -d
    ;;
  "railway")
    echo "🚄 Deploying to Railway..."
    railway up --service voice-marketplace
    ;;
  *)
    echo "❌ Unknown environment: $ENVIRONMENT"
    exit 1
    ;;
esac

echo "✅ Deployment completed!"
echo "🔍 Check status: docker-compose ps"
echo "📋 View logs: docker-compose logs -f voice-marketplace"
echo "🏥 Health check: curl http://localhost:8006/health"
